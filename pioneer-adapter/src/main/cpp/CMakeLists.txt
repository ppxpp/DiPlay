cmake_minimum_required(VERSION 3.10)
project(pioneer_usb C)
add_library(pioneer_usb SHARED pioneer_usb.c)
set(LWIP_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lwip-STABLE-2_2_1_RELEASE)
file(GLOB LWIP_CORE ${LWIP_DIR}/src/core/*.c ${LWIP_DIR}/src/core/ipv6/*.c)
add_library(pioneer_lwip SHARED pioneer_lwip.c ${LWIP_CORE} ${LWIP_DIR}/src/netif/ethernet.c)
target_include_directories(pioneer_lwip PRIVATE lwip-port ${LWIP_DIR}/src/include)
target_compile_options(pioneer_lwip PRIVATE -O2)
add_library(pioneer_crypto SHARED pioneer_crypto.c vendor/monocypher/monocypher.c)
target_include_directories(pioneer_crypto PRIVATE vendor/monocypher)
target_compile_options(pioneer_crypto PRIVATE -O3)
